    if let Err(msg) = auth::requer_permissao(claims, permissao) {
        Err(erro_padrao(
            "FORBIDDEN",
            format!("Permissão negada: {} ({})", permissao, msg),

        Ok(())

#[macro_export]
macro_rules! require_perm {

## 5. Endurecimento do `requer_permissao` (próxima sprint)

```rust
// Próxima sprint — implementar de verdade
pub fn requer_permissao(claims: &Claims, permissao: &str) -> Result<(), String> {
    // 1. SUPER_ADMIN bypass
    if claims.roles.iter().any(|r| r == "SUPER_ADMIN") {
        return Ok(());
    }

    // 2. Verifica se permission está nos claims
    if claims.permissions.iter().any(|p| p == permissao) {
        return Ok(());
    }

    // 3. Fallback: consultar PermissionCache (P2.6.1 trait)
    // (chamada bloqueante em runtime)

    Err(format!("permission '{}' não concedida", permissao))
}
```

**Dependências**:
- `Claims.permissions: Vec<String>` (já existe em P2.6.1)
- `Claims.roles: Vec<String>` (já existe em P2.6.1)
- `PermissionCache` trait (já existe em `rbac::cache`)
